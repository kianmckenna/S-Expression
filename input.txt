(def sumTo (num)
    (if
        (eq? num 0)
        0
        (add num (sumTo (sub num 1)))
    )
)

(sumTo 10)

(def length (list)
    (if
       (nil? list)
       0
       (add 1 (length (cdr list))) 
    )
)

(length '(a b c d e f g h i j k l m n o p q r s t u v w x y z))

(def sumList (list)
    (if
        (nil? list)
        0
        (add (car list) (sumList (cdr list)))        
    )
)

(sumList '(1 2 3 4))

(def contains? (item list)
    (if
        (nil? list)
        ()
        (if 
            (eq? (car list) item)
            'T
            (contains? item (cdr list))
        )
    )
)

(set ex '(a b c d))

(contains? a ex)

(contains? l ex)

(def count (item list)
    (if
        (nil? list)
        0
        (if
            (eq? item (car list))
            (add 1 (count item (cdr list))) 
            (count item (cdr list))  
        )
    )
)

(count 'a '(a b c d a a a a b))

(def construct (list)
    (if
        (nil? list)
        ()
        (cons (car list) (construct (cdr list)))
    )
)

(construct '(a b c))

(def append (list1 list2)
    (if
        (and? (nil? list1) (nil? list2))
        ()
        (if
            (nil? list1)
            (if
                (nil? list2)
                ()
                (cons
                    (car list2)
                    (append list1 (cdr list2))
                )
            )
            (cons
                (car list1)
                (append (cdr list1) list2)
            )
        )
    )
)

(append '(a b) '(c d e))

(def factorial (num)
    (if
        (eq? num 0)
        1
        (if
            (eq? num 1)
            1
            (mul num (factorial (sub num 1)))
        )
    )
)

(factorial 5)
(factorial 0)

(def allNumbers? (list)
    (if
        (nil? list)
        'T
        (if
            (number? (car list))
            (allNumbers? (cdr list))
            ()
        )
    )
)

(def reverseHelper (list reversed)
    (if
        (nil? list)
        reversed
        (reverseHelper (cdr list) (cons (car list) reversed))
    )
)

(def reverse (list)
    (reverseHelper list '())
)

(reverse '(w x y z))

(allNumbers? '(1 2 3 4 5))
(allNumbers? '(1 2 3 x 5))

(def compare (num1 num2)
    (if
        (lt num1 num2)
        num2
        num1
    )
)

(compare 2 1)
(compare 3 4)

(def max (list)
    (if
        (nil? list)
        ()
        (if
            (nil? (cdr list))
            (car list)
            (compare (car list) (max (cdr list)))
        )
    )
)

(max '(4 9 2 7))
